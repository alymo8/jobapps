% Custom commands

\newcommand{\resumeItem}[1]{
  \item\small{{#1 \vspace{-2pt}}}
}

% Standard heading: Company/School + Date | Role/Degree + Location
\newcommand{\resumeSubheading}[4]{
  \vspace{-2pt}\item
    \begin{tabularx}{1.0\textwidth}[t]{@{}>{\raggedright\arraybackslash}X@{\hspace{8pt}}r@{}}
      \textbf{#1} & \textbf{\small #2} \\
      \textit{\small#3} & \textit{\small #4} \\
    \end{tabularx}\vspace{-7pt}
}

% Continued role at the SAME company (no repeated company name)
% Args: {Role | Tech} {Date range} {Location}
\newcommand{\resumeSubheadingContinued}[3]{
  \vspace{4pt}
    \begin{tabularx}{1.0\textwidth}[t]{@{}>{\raggedright\arraybackslash}X@{\hspace{8pt}}r@{}}
      & \textbf{\small #2} \\
      \textit{\small#1} & \textit{\small #3} \\
    \end{tabularx}\vspace{-7pt}
}

% Additional degree at the SAME institution (no repeated school name)
% Arg: {Degree name}
\newcommand{\resumeSubheadingDegree}[2]{
  \vspace{2pt}
    \begin{tabular*}{1.0\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textit{\small#1} & \textit{\small #2} \\
    \end{tabular*}\vspace{-7pt}
}

    \end{tabular*}\vspace{-7pt}
}
